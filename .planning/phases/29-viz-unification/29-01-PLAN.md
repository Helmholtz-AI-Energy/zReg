  MAX_PTS = 4_000
  COLOR   = "#2a6496"

  def _scatter3(ax, pts, title):
      rng = np.random.default_rng(0)
      if len(pts) > MAX_PTS:
          idx = rng.choice(len(pts), MAX_PTS, replace=False)
          pts = pts[idx]
      ax.scatter(pts[:, 0], pts[:, 1], pts[:, 2], s=1.5, alpha=0.45, c=COLOR, linewidths=0)
      ax.set_title(title, fontsize=9, pad=4)
      for label in (ax.get_xticklabels() + ax.get_yticklabels() + ax.get_zticklabels()):
          label.set_fontsize(6)
      ax.set_xlabel("x", fontsize=7, labelpad=2)
      ax.set_ylabel("y", fontsize=7, labelpad=2)
      ax.set_zlabel("z", fontsize=7, labelpad=2)
      ax.xaxis.pane.fill = False

      ax.zaxis.pane.fill = False

  def _frame_indices(csv_path):
      vals = pd.read_csv(csv_path, usecols=["t"])["t"]
      return sorted(vals.unique())

  def _load_three_frames(csv_path, wanted):
      parts = []
      for chunk in pd.read_csv(csv_path, usecols=["x","y","z","t"], chunksize=60_000):
          sub = chunk[chunk["t"].isin(wanted)]
          if len(sub): parts.append(sub)
      if not parts: return {}
      df = pd.concat(parts, ignore_index=True)
      return {t: df.loc[df["t"] == t, ["x","y","z"]].values for t in wanted}
