From scripts/generate_datasets.py (numpy source to adapt — DO NOT import this script, port the math):

  def in_bowl(pts, R, d):
      # pts: (M, 3) numpy array
      norm_sq  = (pts ** 2).sum(axis=1)
      carve_sq = pts[:, 0] ** 2 + pts[:, 1] ** 2 + (pts[:, 2] - d) ** 2
      return (norm_sq <= R ** 2) & (carve_sq > R ** 2 + d ** 2) & (pts[:, 2] <= 0.0)

  def sample_ball_shell(n, R_inner, R_outer, rng):   # use R_inner=0.0 for a solid ball
      r     = rng.uniform(R_inner ** 3, R_outer ** 3, n) ** (1.0 / 3)
      cos_t = rng.uniform(-1.0, 1.0, n)
      phi   = rng.uniform(0.0, 2.0 * np.pi, n)
      sin_t = np.sqrt(np.clip(1.0 - cos_t ** 2, 0.0, None))
      return np.stack([r*sin_t*np.cos(phi), r*sin_t*np.sin(phi), r*cos_t], axis=1).astype(np.float32)

  def sample_bowl_frame(n, R, d, rng, batch_mult=10):   # rejection sampling
      collected, total = [], 0
      batch = max(n * batch_mult, 2_000)
      while total < n:
          cands = rng.uniform(-R, R, (batch, 3))
          keep  = cands[in_bowl(cands, R, d)]
          collected.append(keep); total += len(keep)
      return np.concatenate(collected, axis=0)[:n].astype(np.float32)
