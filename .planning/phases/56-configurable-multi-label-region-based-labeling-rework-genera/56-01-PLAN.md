import sys; sys.path.insert(0, 'src')
import torch
from zreg.data_generation.labels import LabelComponentSpec, LabelSpec, _component_score, _label_scores

# Point exactly at a blob center scores higher than a point far away.
pos_near = torch.tensor([[0.0, 0.0, 0.0]])
pos_far = torch.tensor([[10.0, 0.0, 0.0]])
blob = LabelComponentSpec(shape='blob', center=[0.0, 0.0, 0.0], sigma=1.0)
assert _component_score(pos_near, blob, 'deterministic').item() > _component_score(pos_far, blob, 'deterministic').item()

# Point at the cone pole scores higher than a point at the equator.
cone = LabelComponentSpec(shape='cone', center=[0.0, 0.0, 1.0], sigma=20.0)
pole_pt = torch.tensor([[0.0, 0.0, 1.0]])
eq_pt = torch.tensor([[1.0, 0.0, 0.0]])
assert _component_score(pole_pt, cone, 'deterministic').item() > _component_score(eq_pt, cone, 'deterministic').item()

# Voronoi deterministic mode: nearer center scores higher; probabilistic requires temperature.
voronoi = LabelComponentSpec(shape='voronoi', center=[5.0, 0.0, 0.0])
assert _component_score(pos_far, voronoi, 'deterministic').item() > _component_score(pos_near, voronoi, 'deterministic').item()
try:
    _component_score(pos_near, voronoi, 'probabilistic')
    print('FAIL: expected ValueError for missing temperature in probabilistic voronoi')
    sys.exit(1)
except ValueError as e:
    assert 'temperature' in str(e)

voronoi_t = LabelComponentSpec(shape='voronoi', center=[5.0, 0.0, 0.0], temperature=2.0)
score = _component_score(pos_near, voronoi_t, 'probabilistic')
assert score.shape == (1,)

# _label_scores: mixture of two components, shape (N,), empty-frame safe.
spec = LabelSpec(label_id=0, components=[blob, cone])
scores = _label_scores(pos_near, spec, 'deterministic')
assert scores.shape == (1,)

empty_pos = torch.zeros((0, 3))
empty_scores = _label_scores(empty_pos, spec, 'deterministic')
assert empty_scores.shape == (0,)

print('_component_score/_label_scores checks passed')
